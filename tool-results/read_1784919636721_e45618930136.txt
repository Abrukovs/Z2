   197→// ЧМ по футболу, Евро, Копа Америка — не каждый год. Добавляем только в годы проведения.
   198→function getSpecialEvents(year: number, month: number): ActiveLeagues {
   199→  const special: ActiveLeagues = {
   200→    football: [],
   201→    basketball: [],
   202→    tennis: [],
   203→    hockey: [],
   204→    esports: [],
   205→  };
   206→  const isSummer = month === 6 || month === 7;
   207→
   208→  // Чемпионат мира по футболу FIFA: 2022, 2026, 2030 (июнь-июль)
   209→  if (isSummer && [2022, 2026, 2030, 2034].includes(year)) {
   210→    special.football.push(
   211→      'Чемпионат мира по футболу',
   212→      'ЧМ по футболу',
   213→      'FIFA World Cup',
   214→      'ЧМ-' + year,
   215→    );
   216→  }
   217→  // Евро УЕФА: 2024, 2028, 2032 (июнь-июль)
   218→  if (isSummer && [2024, 2028, 2032].includes(year)) {
   219→    special.football.push('Чемпионат Европы', 'Евро УЕФА', 'Евро-' + year);
   220→  }
   221→  // Копа Америка: 2024, 2028 (июнь-июль)
   222→  if (isSummer && [2024, 2028].includes(year)) {
   223→    special.football.push('Копа Америка', 'Copa America');
   224→  }
   225→  return special;
   226→}

   228→function getActiveLeagues(year: number, month: number): ActiveLeagues {
   229→  const base =
   230→    SEASONAL_LEAGUES[month] || {
   231→      football: [] as string[],
   232→      basketball: [] as string[],

   234→      hockey: [] as string[],
   235→      esports: [] as string[],
   236→      volleyball: [] as string[],
   237→      baseball: [] as string[],
   238→    };
   239→  const special = getSpecialEvents(year, month);
   240→  // MLB сезон: апрель-октябрь (включая All-Star Game в июле)
   241→  const mlbSeason = month >= 4 && month <= 10;
   242→  return {
   243→    football: [...base.football, ...special.football],
   244→    basketball: [...base.basketball, ...special.basketball],
   245→    tennis: [...base.tennis, ...special.tennis],
   246→    hockey: [...base.hockey, ...special.hockey],
   247→    esports: [...base.esports, ...special.esports],
   248→    volleyball: base.volleyball || [],
   249→    baseball: mlbSeason ? ['MLB'] : [],
   250→  };
   251→}

   253→// Проверка: относится ли строка лиги к разрешённым в текущем месяце.
   254→function isLeagueAllowed(leagueName: string, allowed: string[]): boolean {
   255→  if (allowed.length === 0) return false; // нет активных лиг → всё запрещено
   256→  const lower = (leagueName || '').toLowerCase();
   257→  return allowed.some((a) => lower.includes(a.toLowerCase()));
   258→}

   260→// Какой турнир Большого шлема идёт сейчас?
   261→// Учитывает конкретные даты проведения, а не только месяц.
   262→// Возвращает имя для поискового запроса или null.
   263→function getActiveGrandSlam(
   264→  year: number,
   265→  month: number,
   266→  day: number,
   267→): string | null {
   268→  // Australian Open: 12-26 января
   269→  if (month === 1 && day >= 12 && day <= 27) return 'Australian Open';
   270→  // Roland Garros: 25 мая - 8 июня
   271→  if ((month === 5 && day >= 25) || (month === 6 && day <= 9))
   272→    return 'Roland Garros';
   273→  // Уимблдон: 29 июня - 12 июля
   274→  if ((month === 6 && day >= 29) || (month === 7 && day <= 12)) return 'Уимблдон';
   275→  // US Open: 25 августа - 13 сентября
   276→  if ((month === 8 && day >= 25) || (month === 9 && day <= 14)) return 'US Open';
   277→  return null;
   278→}

   392→  const found = SPORT_MAP[norm];
   393→  if (found) {
   394→    return { label: found.label, icon: found.icon, key: found.key };
   395→  }
   396→  // fallback: найти по частичному совпадению
   397→  const partial = Object.values(SPORT_MAP).find((m) =>
   398→    norm.includes(m.label.toLowerCase()),
   399→  );
   400→  if (partial) {
   401→    return {
   402→      label: partial.label,
   403→      icon: partial.icon,
   404→      key: partial.key,
   405→    };
   406→  }
   407→  return { label: label || 'Спорт', icon: '🏆', key: fallbackKey };
   408→}

   429→// Создать детерминированный ID пика на основе матча.
   430→// Один и тот же матч (sport+teams+time+date) всегда даёт один и тот же ID,
   431→// чтобы повторные генерации не создавали дубликаты в БД статистики.
   432→function makePickId(
   433→  sportKey: string,
   434→  team1: string,
   435→  team2: string,
   436→  startTime: string,
   437→  pickDate = '',
   438→): string {
   439→  const slug = (s: string) =>
   440→    s
   441→      .toLowerCase()
   442→      .replace(/[^a-zа-яё0-9]/gi, '')
   443→      .slice(0, 12);
   444→  const t1 = slug(team1);
   445→  const t2 = slug(team2);
   446→  const t = (startTime || '').replace(/[^0-9]/g, '');
   447→  return `pick-${pickDate}-${sportKey}-${t1}-${t2}-${t}`;
   448→}

   450→function parsePicks(content: string, generatedAt: string, pickDate = ''): Pick[] {
   451→  let text = content.trim();
   452→
   453→  // Убираем markdown-ограждения
   454→  const fence = text.match(/```(?:json)?\s*([\s\S]*?)```/i);
   455→  if (fence) {
   456→    text = fence[1].trim();
   457→  }
   458→
   459→  // Извлекаем первый JSON-массив
   460→  const start = text.indexOf('[');
   461→  const end = text.lastIndexOf(']');
   462→  if (start !== -1 && end !== -1 && end > start) {
   463→    text = text.slice(start, end + 1);
   464→  }
   465→
   466→  let arr: unknown;
   467→  try {
   468→    arr = JSON.parse(text);
   469→  } catch {
   470→    return [];
   471→  }
   472→
   473→  if (!Array.isArray(arr)) return [];
   474→
   475→  const picks: Pick[] = arr
   476→    .map((item, idx): Pick | null => {
   477→      const r = item as RawPick;
   478→      if (!r || typeof r !== 'object') return null;
   479→      if (!r.team1 || !r.team2) return null;
   480→
   481→      const sportLabel = r.sport || 'Спорт';
   482→      const meta = sportMeta(sportLabel, `sport-${idx}`);
   483→      const conf = normalizeConfidence(r.confidence);
   484→      const score = normalizeScore(r.confidenceScore, conf);
   485→
   486→      return {
   487→        id: makePickId(meta.key, String(r.team1 || ''), String(r.team2 || ''), normalizeTime(r.time), pickDate),
   488→        sport: meta.label,
   489→        sportKey: meta.key,
   490→        sportIcon: meta.icon,
   491→        league: r.league || '—',
   492→        team1: String(r.team1).trim(),
   493→        team2: String(r.team2).trim(),
   494→        startTime: normalizeTime(r.time),
   495→        market: r.market || 'Исход матча',
   496→        selection: r.selection || '—',
   497→        odds: normalizeOdds(r.odds),
   498→        confidence: conf,
   499→        confidenceScore: score,
   500→        reasoning: (r.reasoning || '').trim(),
   501→        risk: (r.risk || '').trim(),
   502→        sourceQuote: (r.sourceQuote || '').trim(),
   503→        sourceUrl: (r.sourceUrl || '').trim(),
   504→        sourceName: (r.sourceName || '').trim(),
   505→        dataSource: (r.dataSource || 'Веб-поиск').trim(),
   506→        generatedAt,
   507→      };
   508→    })
   509→    .filter((p): p is Pick => p !== null);
   510→
   511→  return picks;
   512→}

   514→function normalizeConfidence(v?: string): Confidence {
   515→  const s = (v || '').toLowerCase().trim();
   516→  if (s.startsWith('high') || s.startsWith('выс') || s === 'высокая')
   517→    return 'high';
   518→  if (s.startsWith('low') || s.startsWith('низ') || s === 'низкая')
   519→    return 'low';
   520→  return 'medium';
   521→}
   522→
   523→function normalizeScore(v: unknown, conf: Confidence): number {
   524→  let n = typeof v === 'number' ? v : parseFloat(String(v ?? ''));
   525→  if (!isFinite(n) || n <= 0) {
   526→    n = conf === 'high' ? 82 : conf === 'medium' ? 62 : 42;
   527→  }
   528→  if (n > 0 && n <= 1) n = Math.round(n * 100); // 0.8 -> 80
   529→  return Math.max(5, Math.min(99, Math.round(n)));
   530→}
   531→
   532→function normalizeTime(v?: string): string {
   533→  if (!v) return '';
   534→  const s = String(v).trim();
   535→  const m = s.match(/(\d{1,2}):(\d{2})/);
   536→  if (m) {
   537→    return `${m[1].padStart(2, '0')}:${m[2]}`;
   538→  }
   539→  return s.slice(0, 5);
   540→}
   541→
   542→function normalizeOdds(v: unknown): string {
   543→  let n = typeof v === 'number' ? v : parseFloat(String(v ?? ''));
   544→  if (!isFinite(n) || n <= 1) return '—';
   545→  return n.toFixed(2).replace('.', '.');
   546→}

   548→async function searchWithRetry(
   549→  zai: Awaited<ReturnType<typeof ZAI.create>>,
   550→  query: string,
   551→  num: number,
   552→  retries = 3,
   553→): Promise<unknown[]> {
   554→  let lastError: unknown;
   555→  for (let attempt = 1; attempt <= retries; attempt++) {
   556→    try {
   557→      // recency_days=1 — предпочитаем материалы за последние сутки,
   558→      // чтобы отсечь старые новости о прошедших матчах
   559→      const res = await zai.functions.invoke('web_search', {
   560→        query,
   561→        num,
   562→        recency_days: 1,
   563→      });
   564→      bumpWebSearch();
   565→      return (res as unknown[]) || [];
   566→    } catch (e) {
   567→      lastError = e;
   568→      const msg = e instanceof Error ? e.message : String(e);
   569→      const isRateLimit = msg.includes('429') || msg.includes('Too many');
   570→      if (attempt < retries) {
   571→        const delay = isRateLimit ? 1500 * attempt : 800 * attempt;
   572→        await new Promise((r) => setTimeout(r, delay));
   573→      }
   574→    }
   575→  }
   576→  throw lastError;
   577→}

   579→function todayRu(): {
   580→  iso: string;
   581→  pretty: string;
   582→  weekday: string;
   583→  ymd: string;
   584→  timeMsk: string;
   585→  year: number;
   586→  month: number;
   587→  day: number;
   588→  monthName: string;
   589→} {
   590→  const now = new Date();
   591→  const pretty = now.toLocaleDateString('ru-RU', {
   592→    day: 'numeric',
   593→    month: 'long',
   594→    year: 'numeric',
   595→    timeZone: 'Europe/Moscow',
   596→  });
   597→  const weekday = now.toLocaleDateString('ru-RU', {
   598→    weekday: 'long',
   599→    timeZone: 'Europe/Moscow',
   600→  });
   601→  const monthName = now.toLocaleDateString('ru-RU', {
   602→    month: 'long',
   603→    timeZone: 'Europe/Moscow',
   604→  });
   605→  // Локальные дата/время по Мск в формате YYYY-MM-DD и HH:MM
   606→  const mskParts = new Intl.DateTimeFormat('en-CA', {
   607→    timeZone: 'Europe/Moscow',
   608→    year: 'numeric',
   609→    month: '2-digit',
   610→    day: '2-digit',
   611→    hour: '2-digit',
   612→    minute: '2-digit',
   613→    hour12: false,
   614→  }).formatToParts(now);
   615→  const get = (t: string) =>
   616→    mskParts.find((p) => p.type === t)?.value || '';
   617→  const ymd = `${get('year')}-${get('month')}-${get('day')}`;
   618→  const timeMsk = `${get('hour')}:${get('minute')}`;
   619→  const year = parseInt(get('year'), 10);
   620→  const month = parseInt(get('month'), 10);
   621→  const day = parseInt(get('day'), 10);
   622→  return { iso: now.toISOString(), pretty, weekday, ymd, timeMsk, year, month, day, monthName };
   623→}

   657→function hasPastMarker(text: string): { hit: boolean; marker: string | null } {
   658→  const lower = (text || '').toLowerCase();
   659→  for (const m of PAST_MARKERS) {
   660→    if (lower.includes(m)) {
   661→      return { hit: true, marker: m };
   662→    }
   663→  }
   664→  return { hit: false, marker: null };
   665→}

   667→// Проверка: время начала уже прошло по Мск?
   668→function timeAlreadyPassed(startTime: string, currentTimeMsk: string): boolean {
   669→  if (!startTime || !currentTimeMsk) return false;
   670→  const sm = startTime.match(/^(\d{1,2}):(\d{2})$/);
   671→  const cm = currentTimeMsk.match(/^(\d{1,2}):(\d{2})$/);
   672→  if (!sm || !cm) return false;
   673→  const sMin = parseInt(sm[1], 10) * 60 + parseInt(sm[2], 10);
   674→  const cMin = parseInt(cm[1], 10) * 60 + parseInt(cm[2], 10);
   675→  // Если матч начинается более чем на 30 минут раньше текущего времени — считаем прошедшим.
   676→  // (даём буфер, чтобы не отсечь только что начавшиеся)
   677→  return sMin + 30 < cMin;
   678→}

   698→  const cacheKey = `picks-${dayKey}`;
   699→  const active = getActiveLeagues(year, month);
   700→
   701→  const cached = cache.get(cacheKey);
   702→  if (cached && !opts.force && Date.now() - cached.ts < CACHE_TTL) {
   703→    usage.cacheHits += 1;
   704→    // При отдаче из кэша отсекаем уже начавшиеся матчи.
   705→    // Пики сгенерированы, скажем, в 10:00, а сейчас 18:00 — часть матчей уже прошла.
   706→    // Не показываем их (статистику по ним сохраняем как есть, просто не отображаем).
   707→    const livePicks = cached.picks.filter((p) => {
   708→      // Для ESPN-пиков не применяем HH:MM-фильтр (они валидированы по UTC-дате).
   709→      // ESPN-матч может быть в 01:40 МСК (ночь следующего дня) — HH:MM-сравнение
   710→      // лочно посчитает его прошедшим.
   711→      if (p.dataSource === 'ESPN API') return true;
   712→      if (!p.startTime || !timeMsk) return true;

   716→      picks: livePicks,

  1076→    baseball: active.baseball || ['MLB'],
  1077→  };
  1078→  const picks: Pick[] = [];
  1079→  let filteredOut = 0;
  1080→  for (const p of allPicks) {
  1081→    const allowed = allowedBySport[p.sportKey] || [];
  1082→    const isFromApi = p.dataSource === 'ESPN API' || p.dataSource === 'API-Sports';
  1083→
  1084→    // Проверка 0 (ВАЛИДАЦИЯ СПОРТ↔ЛИГА): отлов mismatches вида «Футбол + Уимблдон».
  1085→    // Если в лиге явно указан турнир другого вида спорта — отбраковываем.
  1086→    const leagueLower = (p.league || '').toLowerCase();
  1087→    const WRONG_LEAGUE_FOR_SPORT: Record<string, string[]> = {
  1088→      football: ['уимблдон', 'roland garros', 'australian open', 'us open', 'wnba', 'mlb', 'nba', 'нхл', 'кхл', 'iem', 'lol', 'лига наций волейбол'],
  1089→      basketball: ['уимблдон', 'roland garros', 'australian open', 'us open', 'mlb', 'чемпионат мира по футболу', 'чм-2026', 'апл', 'ла лига', 'iem', 'лига наций волейбол'],
  1090→      tennis: ['wnba', 'mlb', 'nba', 'нхл', 'кхл', 'чемпионат мира по футболу', 'чм-2026', 'апл', 'iem', 'lol', 'лига наций волейбол'],
  1091→      hockey: ['уимблдон', 'wnba', 'mlb', 'nba', 'чемпионат мира по футболу', 'iem', 'лига наций волейбол'],
  1092→      esports: ['уимблдон', 'roland garros', 'wnba', 'mlb', 'nba', 'нхл', 'кхл', 'чемпионат мира по футболу', 'апл', 'ла лига', 'лига наций волейбол'],
  1093→      baseball: ['уимблдон', 'roland garros', 'wnba', 'nba', 'нхл', 'кхл', 'чемпионат мира по футболу', 'апл', 'iem', 'лига наций волейбол'],
  1094→      volleyball: ['уимблдон', 'roland garros', 'wnba', 'mlb', 'nba', 'нхл', 'кхл', 'чемпионат мира по футболу', 'апл', 'iem'],
  1095→    };
  1096→    const wrongLeagues = WRONG_LEAGUE_FOR_SPORT[p.sportKey] || [];
  1097→    const isWrongLeague = wrongLeagues.some((wl) => leagueLower.includes(wl));
  1098→    if (isWrongLeague) {
  1099→      filteredOut += 1;
  1100→      console.warn(
  1101→        `[picks] filtered out (wrong league for sport): sport=${p.sportKey}, league=${p.league} (${p.team1} vs ${p.team2})`,
  1102→      );
  1103→      continue;
  1104→    }
  1105→
  1106→    // Проверка 0b (АНТИ-УИМБЛДОН-ПОСЛЕ-ЗАВЕРШЕНИЯ):
  1107→    // Если БШ-турнир НЕ активен сейчас, а лига содержит имя БШ — это галлюцинация
  1108→    // (LLM нашла архив результатов и лепит БШ-имя на текущий ATP-матч).
  1109→    // Отбраковываем такие пики.
  1110→    const SLAM_NAMES = ['уимблдон', 'wimbledon', 'roland garros', 'australian open', 'us open'];
  1111→    const leagueHasSlam = SLAM_NAMES.some((s) => leagueLower.includes(s));
  1112→    if (leagueHasSlam && !activeSlam) {
  1113→      filteredOut += 1;
  1114→      console.warn(
  1115→        `[picks] filtered out (slam league when no slam active): league=${p.league} (${p.team1} vs ${p.team2}) — slam=${activeSlam || 'NONE'}`,
  1116→      );
  1117→      continue;
  1118→    }
  1119→
  1120→    // Проверка 1: лига должна быть активной в текущем сезоне
  1121→    // (для ESPN-матчей ЧМ/MLB/WNBA это автоматически проходит, т.к. они уже
  1122→    // отфильтрованы по сезонности на этапе getActiveLeagues + ESPN endpoint)
  1123→    if (!isLeagueAllowed(p.league, allowed) && !isFromApi) {
  1124→      filteredOut += 1;
  1125→      console.warn(
  1126→        `[picks] filtered out off-season league: ${p.sport}/${p.league} (${p.team1} vs ${p.team2})`,
  1127→      );
  1128→      continue;
  1129→    }
  1130→
  1131→    // Проверка 2 (АНТИ-ГАЛЛЮЦИНАЦИЯ + АНТИ-CROSS-CONTAMINATION):
  1132→    // Для веб-поиска: sourceQuote обязан содержать оба имени РЯДОМ (≤60 символов).
  1133→    // Для ESPN API-матчей: цитата формируется программно, проверка не нужна.
  1134→    if (!isFromApi) {
  1135→      const quote = (p.sourceQuote || '').toLowerCase().trim();
  1136→      const t1 = (p.team1 || '').toLowerCase().trim();
  1137→      const t2 = (p.team2 || '').toLowerCase().trim();
  1138→      const t1Parts = t1.split(/\s+/).filter((s) => s.length >= 4);
  1139→      const t2Parts = t2.split(/\s+/).filter((s) => s.length >= 4);
  1140→
  1141→      const t1Positions: number[] = [];
  1142→      const t2Positions: number[] = [];
  1143→      for (const part of t1Parts) {
  1144→        let idx = quote.indexOf(part);
  1145→        while (idx !== -1) {
  1146→          t1Positions.push(idx);
  1147→          idx = quote.indexOf(part, idx + 1);
  1148→        }
  1149→      }
  1150→      for (const part of t2Parts) {
  1151→        let idx = quote.indexOf(part);
  1152→        while (idx !== -1) {
  1153→          t2Positions.push(idx);
  1154→          idx = quote.indexOf(part, idx + 1);
  1155→        }
  1156→      }
  1157→
  1158→      const bothPresent = t1Positions.length > 0 && t2Positions.length > 0;
  1159→      let minDistance = Infinity;
  1160→      if (bothPresent) {
  1161→        for (const a of t1Positions) {
  1162→          for (const b of t2Positions) {
  1163→            minDistance = Math.min(minDistance, Math.abs(a - b));
  1164→          }
  1165→        }
  1166→      }
  1167→      const namesClose = minDistance <= 60;
  1168→
  1169→      if (!quote || quote.length < 15 || !bothPresent || !namesClose) {
  1170→        filteredOut += 1;
  1171→        console.warn(
  1172→          `[picks] filtered out (no valid sourceQuote or names too far apart: dist=${minDistance}): ${p.sport}/${p.league} (${p.team1} vs ${p.team2}) — quote="${p.sourceQuote?.slice(0, 80)}"`,
  1173→        );
  1174→        continue;
  1175→      }
  1176→    } else {
  1177→      // Для ESPN-матчей: источник уже валиден, но проверим что пик
  1178→      // соответствует одному из полученных из API событий.
  1179→      // Используем мягкое совпадение: хотя бы по 4 символам каждого имени.
  1180→      const norm = (s: string) => s.toLowerCase().replace(/[^a-zа-яё0-9]/gi, '');
  1181→      const p1 = norm(p.team1);
  1182→      const p2 = norm(p.team2);
  1183→      const matchInApi = apiEvents.some((ev) => {
  1184→        const e1 = norm(ev.team1);
  1185→        const e2 = norm(ev.team2);
  1186→        // Проверяем в обе стороны (home/away могут быть перепутаны)
  1187→        const direct =
  1188→          (e1.includes(p1.slice(0, 4)) || p1.includes(e1.slice(0, 4))) &&
  1189→          (e2.includes(p2.slice(0, 4)) || p2.includes(e2.slice(0, 4)));
  1190→        const swapped =
  1191→          (e2.includes(p1.slice(0, 4)) || p1.includes(e2.slice(0, 4))) &&
  1192→          (e1.includes(p2.slice(0, 4)) || p2.includes(e1.slice(0, 4)));
  1193→        return direct || swapped;
  1194→      });
  1195→      if (!matchInApi) {
  1196→        filteredOut += 1;
  1197→        console.warn(
  1198→          `[picks] filtered out (ESPN pick not in API events): ${p.sport}/${p.league} (${p.team1} vs ${p.team2})`,
  1199→        );
  1200→        continue;
  1201→      }
  1202→    }
  1203→
  1204→    // Проверка 3 (АНТИ-ПРОШЕДШИЕ): если в sourceQuote или reasoning
  1205→    // есть слова-маркеры прошедшего матча («выиграла», «завершился» и т.п.)
  1206→    // — отбраковываем. Это решает случай «Team Falcons выиграла IEM Major».
  1207→    const pastCheck = hasPastMarker(p.sourceQuote + ' ' + p.reasoning + ' ' + p.risk);
  1208→    if (pastCheck.hit) {
  1209→      filteredOut += 1;
  1210→      console.warn(
  1211→        `[picks] filtered out (past event, marker="${pastCheck.marker}"): ${p.sport}/${p.league} (${p.team1} vs ${p.team2}) — quote="${p.sourceQuote?.slice(0, 80)}"`,
  1212→      );
  1213→      continue;
  1214→    }
  1215→
  1216→    // Проверка 3b: если в sourceQuote есть счёт матча (паттерн «цифра:цифра»
  1217→    // или «цифра - цифра» рядом с командами) — это уже сыгранный матч.
  1218→    {
  1219→      // Исключаем время HH:MM (например "13:00"). Убираем без \b, чтобы чистить
  1220→      // и слитые с годом/датой варианты вида "03.07.2611:00" (= дата 03.07.26 + 11:00).
  1221→      // Время всегда имеет 2 цифры после двоеточия; счёт часто с 1 цифрой (2:1).
  1222→      const withoutTime = p.sourceQuote.replace(/\d{1,2}:\d{2}/g, '');
  1223→      if (/\d{1,2}\s*[:\-–]\s*\d{1,2}/.test(withoutTime)) {
  1224→        filteredOut += 1;
  1225→        console.warn(
  1226→          `[picks] filtered out (has score in quote = past match): ${p.sport}/${p.league} (${p.team1} vs ${p.team2}) — quote="${p.sourceQuote?.slice(0, 80)}"`,
  1227→        );
  1228→        continue;
  1229→      }
  1230→    }
  1231→
  1232→    // Проверка 4: если startTime задано и уже прошло по Мск (с буфером 30 мин) — отбраковываем.
  1233→    // ВАЖНО: для ESPN-пиков эту проверку пропускаем — они уже отфильтрованы в filterUpcoming
  1234→    // по полной UTC-дате. ESPN-матч может быть в 01:40 МСК (ночь следующего дня), и простое
  1235→    // сравнение HH:MM ложно посчитает его прошедшим.
  1236→    if (!isFromApi && timeAlreadyPassed(p.startTime, timeMsk)) {
  1237→      filteredOut += 1;
  1238→      console.warn(
  1239→        `[picks] filtered out (time already passed: start=${p.startTime}, now=${timeMsk}): ${p.sport}/${p.league} (${p.team1} vs ${p.team2})`,
  1240→      );
  1241→      continue;
  1242→    }
  1243→
  1244→    // Если LLM не выдала коэффициент — ставим нейтральный дефолт
  1245→    if (!p.odds || p.odds === '—') {
  1246→      p.odds = '1.90';
  1247→    }
  1248→    picks.push(p);
  1249→  }
  1250→
  1251→  // Дедупликация: если несколько пиков на один матч (team1+team2+startTime),
  1252→  // оставляем только первый.
  1253→  const seen = new Set<string>();
  1254→  const deduped: Pick[] = [];
  1255→  for (const p of picks) {
  1256→    const key = `${p.sportKey}|${p.team1.toLowerCase()}|${p.team2.toLowerCase()}|${p.startTime}`.toLowerCase();
  1257→    if (seen.has(key)) {
  1258→      filteredOut += 1;
  1259→      continue;
  1260→    }
  1261→    seen.add(key);
  1262→    deduped.push(p);
  1263→  }
  1264→
  1265→  if (deduped.length > 0) {
  1266→    cache.set(cacheKey, { picks: deduped, ts: Date.now() });
  1267→    usage.generations += 1;
  1268→  }
  1269→
  1270→  // Сохраняем пики в БД для ведения статистики (не блокируя ответ).
  1271→  // Это бесплатно — никаких API-вызовов, только локальная SQLite.
  1272→  savePicksToDb(deduped, dayKey).catch((e) =>
  1273→    console.error('[picks] savePicksToDb failed:', e),
  1274→  );
  1275→
  1276→  return {
  1277→    picks: deduped,
  1278→    cached: false,
  1279→    date: dayKey,
  1280→    usage: getUsage(),
  1281→    activeLeagues: active,
  1282→    sportsSearched: SPORTS.length,
  1283→    filteredOut,
  1284→  };
  1285→}
